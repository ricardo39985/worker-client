cmake_minimum_required(VERSION 3.25)
project(OrganizerWorker VERSION 0.1.2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
include(CheckIPOSupported)
check_ipo_supported(RESULT OW_HAS_IPO OUTPUT OW_IPO_ERROR)
option(OW_ENABLE_IPO "Enable checked release link-time optimization" ON)
if(OW_ENABLE_IPO AND OW_HAS_IPO)
  set(CMAKE_INTERPROCEDURAL_OPTIMIZATION_RELEASE ON)
endif()
option(OW_SANITIZE "Enable ASan and UBSan for local tests" OFF)
set(OW_BOOST_ROOT "" CACHE PATH "Verified Boost source root (release uses 1.89.0)")
if(OW_BOOST_ROOT)
  if(NOT EXISTS "${OW_BOOST_ROOT}/boost/json.hpp")
    message(FATAL_ERROR "Boost.JSON headers missing. Run scripts/setup.ps1; do not guess include paths.")
  endif()
  set(OW_BOOST_INCLUDE "${OW_BOOST_ROOT}")
else()
  find_path(OW_BOOST_INCLUDE boost/json/src.hpp REQUIRED)
  message(WARNING "Using installed Boost headers for development only; release builds use the locked archive.")
endif()
find_package(Threads REQUIRED)
add_library(ow_json STATIC src/boost_json.cpp)
set_property(TARGET ow_json PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE OFF)
target_include_directories(ow_json SYSTEM PUBLIC "${OW_BOOST_INCLUDE}")
target_compile_definitions(ow_json PUBLIC BOOST_JSON_NO_LIB BOOST_ALL_NO_LIB)
add_library(ow_core STATIC src/admission.cpp src/store.cpp src/protocol.cpp src/pairing.cpp)
target_include_directories(ow_core PUBLIC include)
target_link_libraries(ow_core PUBLIC ow_json Threads::Threads)
if(WIN32)
  target_compile_definitions(ow_core PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
  target_link_libraries(ow_core PUBLIC winsqlite3)
else()
  find_package(SQLite3 3.22 REQUIRED)
  target_link_libraries(ow_core PUBLIC SQLite::SQLite3)
endif()
if(MSVC)
  target_compile_options(ow_core PRIVATE /W4 /permissive- /utf-8 /EHsc)
else()
  target_compile_options(ow_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
endif()
if(OW_SANITIZE AND NOT MSVC)
  target_compile_options(ow_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(ow_core PUBLIC -fsanitize=address,undefined)
endif()
include(CTest)
if(BUILD_TESTING)
 add_executable(ow_contract_tests tests/main.cpp tests/admission_test.cpp tests/store_test.cpp tests/protocol_test.cpp tests/pairing_test.cpp)
 target_include_directories(ow_contract_tests PRIVATE tests)
 target_link_libraries(ow_contract_tests PRIVATE ow_core)
 add_test(NAME worker_contracts COMMAND ow_contract_tests)
 set_tests_properties(worker_contracts PROPERTIES TIMEOUT 30)
 add_executable(ow_rendition_contract_tests tests/main.cpp tests/media_rendition_test.cpp)
 target_include_directories(ow_rendition_contract_tests PRIVATE include tests)
 add_test(NAME rendition_contracts COMMAND ow_rendition_contract_tests)
 set_tests_properties(rendition_contracts PROPERTIES TIMEOUT 30)
 add_executable(ow_rendition_probe tests/media_rendition_probe.cpp)
 target_include_directories(ow_rendition_probe PRIVATE include)
 add_executable(ow_progress_contract_tests tests/progress_test.cpp)
 target_include_directories(ow_progress_contract_tests PRIVATE include)
 target_link_libraries(ow_progress_contract_tests PRIVATE Threads::Threads)
 add_test(NAME progress_contracts COMMAND ow_progress_contract_tests)
 set_tests_properties(progress_contracts PROPERTIES TIMEOUT 30)
endif()
if(WIN32)
 add_library(ow_windows STATIC windows/platform.cpp windows/process.cpp windows/http.cpp windows/worker.cpp windows/pairing_adapter.cpp)
 target_include_directories(ow_windows PUBLIC windows)
 target_link_libraries(ow_windows PUBLIC ow_core winhttp bcrypt crypt32 shell32 advapi32 user32 ole32)
 if(MSVC)
   target_compile_options(ow_windows PRIVATE /W4 /permissive- /utf-8 /EHsc)
 endif()
 add_executable(OrganizerWorker WIN32 windows/tray.cpp)
 target_link_libraries(OrganizerWorker PRIVATE ow_windows)
 add_executable(OrganizerWorkerConsole windows/console.cpp)
 target_link_libraries(OrganizerWorkerConsole PRIVATE ow_windows)
 add_executable(OrganizerWorkerDoctor windows/doctor.cpp)
 target_link_libraries(OrganizerWorkerDoctor PRIVATE ow_windows)
 if(BUILD_TESTING)
   add_executable(ow_test_child tests/windows_child.cpp)
   target_link_libraries(ow_test_child PRIVATE ow_windows)
   add_executable(ow_windows_tests tests/main.cpp tests/windows_test.cpp)
   target_include_directories(ow_windows_tests PRIVATE tests)
   target_link_libraries(ow_windows_tests PRIVATE ow_windows)
   add_dependencies(ow_windows_tests ow_test_child)
   add_test(NAME windows_contracts COMMAND ow_windows_tests)
   add_test(NAME setup_contracts COMMAND powershell.exe -NoLogo -NoProfile -ExecutionPolicy Bypass -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/setup_policy_test.ps1")
   set_tests_properties(setup_contracts PROPERTIES TIMEOUT 60)
   add_test(NAME setup_progress_contracts COMMAND powershell.exe -NoLogo -NoProfile -ExecutionPolicy Bypass -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/setup_progress_test.ps1")
   set_tests_properties(setup_progress_contracts PROPERTIES TIMEOUT 90)
   set_tests_properties(windows_contracts PROPERTIES TIMEOUT 60)
 endif()
 install(TARGETS OrganizerWorker OrganizerWorkerConsole OrganizerWorkerDoctor RUNTIME DESTINATION .)
endif()
