cmake_minimum_required(VERSION 3.25)
project(OrganizerWorker VERSION 0.1.2 LANGUAGES CXX)
execute_process(COMMAND git rev-parse --short=12 HEAD WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" OUTPUT_VARIABLE OW_BUILD_REVISION OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
if(NOT OW_BUILD_REVISION)
 set(OW_BUILD_REVISION "source-archive")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
include(CheckIPOSupported)
check_ipo_supported(RESULT OW_HAS_IPO OUTPUT OW_IPO_ERROR)
option(OW_ENABLE_IPO "Enable checked release link-time optimization" ON)
if(OW_ENABLE_IPO AND OW_HAS_IPO)
  set(CMAKE_INTERPROCEDURAL_OPTIMIZATION_RELEASE ON)
endif()
option(OW_SANITIZE "Enable ASan and UBSan for local tests" OFF)
set(OW_BOOST_ROOT "" CACHE PATH "Verified Boost source root (release uses 1.89.0)")
if(OW_BOOST_ROOT)
  if(NOT EXISTS "${OW_BOOST_ROOT}/boost/json.hpp")
    message(FATAL_ERROR "Boost.JSON headers missing. Run scripts/setup.ps1; do not guess include paths.")
  endif()
  set(OW_BOOST_INCLUDE "${OW_BOOST_ROOT}")
else()
  find_path(OW_BOOST_INCLUDE boost/json/src.hpp REQUIRED)
  message(WARNING "Using installed Boost headers for development only; release builds use the locked archive.")
endif()
find_package(Threads REQUIRED)
add_library(ow_json STATIC src/boost_json.cpp)
set_property(TARGET ow_json PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE OFF)
target_include_directories(ow_json SYSTEM PUBLIC "${OW_BOOST_INCLUDE}")
target_compile_definitions(ow_json PUBLIC BOOST_JSON_NO_LIB BOOST_ALL_NO_LIB)
add_library(ow_core STATIC src/admission.cpp src/store.cpp src/protocol.cpp src/pairing.cpp src/adaptive_hardware.cpp src/control.cpp)
target_include_directories(ow_core PUBLIC include)
target_link_libraries(ow_core PUBLIC ow_json Threads::Threads)
if(WIN32)
  target_compile_definitions(ow_core PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
  target_link_libraries(ow_core PUBLIC winsqlite3)
else()
  find_package(SQLite3 3.22 REQUIRED)
  target_link_libraries(ow_core PUBLIC SQLite::SQLite3)
endif()
if(MSVC)
  target_compile_options(ow_core PRIVATE /W4 /permissive- /utf-8 /EHsc)
else()
  target_compile_options(ow_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
endif()
if(OW_SANITIZE AND NOT MSVC)
  target_compile_options(ow_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(ow_core PUBLIC -fsanitize=address,undefined)
endif()

   target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
   target_link_options(${target} PRIVATE -fsanitize=address,undefined)
