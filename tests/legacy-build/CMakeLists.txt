cmake_minimum_required(VERSION 3.25)
project(LegacyWorkerAcceptance LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT EXISTS "${OW_LEGACY_SOURCE}/windows/worker.cpp")
 message(FATAL_ERROR "Exact archived legacy source required")
endif()
set(BUILD_TESTING OFF CACHE BOOL "No unrelated legacy test build" FORCE)
add_subdirectory("${OW_LEGACY_SOURCE}" legacy)
add_executable(ow_legacy_actor ../windows_legacy_actor.cpp)
target_compile_definitions(ow_legacy_actor PRIVATE OW_LEGACY_ACTOR=1)
target_link_libraries(ow_legacy_actor PRIVATE ow_windows)
